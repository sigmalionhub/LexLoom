# Crawling is allowed here on purpose, and that is not a contradiction.
#
# Every response from this origin carries `X-Robots-Tag: noindex`, and the page
# itself carries `<meta name="robots" content="noindex">`. Those are the
# directives that remove a page from a search index — but a crawler only obeys
# what it is allowed to fetch. `Disallow: /` would stop it reading them, so a URL
# discovered from a link elsewhere would stay indexed, listed without a snippet,
# and could not be removed. Forbidding the crawl preserves the indexing it looks
# like it prevents.
User-agent: *
Disallow:
